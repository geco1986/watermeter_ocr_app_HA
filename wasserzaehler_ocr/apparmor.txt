#include <tunables/global>

# AppArmor-Profil für das Wasserzähler-OCR-Add-on.
# Das bloße Vorhandensein eines gültigen Profils hebt die
# Home-Assistant-Sicherheitsbewertung um +1. Das Profil ist bewusst
# funktionsfähig gehalten (nicht maximal restriktiv), damit alle
# Laufzeit-Aktionen des Add-ons weiter erlaubt sind: der optionale
# Ollama-Download und -Start, curl/tar/zstd/cpulimit, Python und der
# Schreibzugriff auf den persistenten /data-Speicher.
#
# WICHTIG: Startet das Add-on nach dem Hinzufügen nicht mehr, ist fast
# immer dieses Profil zu streng. Dann diese Datei entfernen (die Bewertung
# fällt wieder auf 7) oder die betroffene Regel lockern.

profile wasserzaehler_ocr flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Netzwerk (HA-Core-API, Ollama, Cloud-OCR-Anbieter, Ollama-Download)
  network,

  # Capabilities (u. a. Signale/Steuerung der Kindprozesse, cpulimit)
  capability,
  signal,

  # Genereller Dateizugriff (lesen/schreiben); Ausführung wird unten
  # gezielt über 'ix' erlaubt.
  file,

  # Entrypoint, Shell und Interpreter
  /usr/bin/env ix,
  /bin/bash ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /app/run.sh ix,

  # Python-Laufzeit und App-Code (nur lesen)
  /usr/local/lib/python3.11/** r,
  /app/** r,

  # Persistenter Add-on-Speicher: lesen/schreiben, und die dort zur
  # Laufzeit abgelegte Ollama-Binary ausführen dürfen
  /data/** rwk,
  /data/ollama/** ix,
  /data/ollama_models/** rwk,

  # Temporärdateien (Ollama-Download, Bildpuffer)
  /tmp/** rwk,

  # System-/Prozessinfos, die die CPU-/RAM-Anzeige liest
  /proc/** r,
  /sys/** r,
}
